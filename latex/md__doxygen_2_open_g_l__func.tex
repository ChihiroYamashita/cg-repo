\begin{DoxyEnumerate}
\item {\bfseries{{\ttfamily GL\+\_\+\+MODELVIEW}（モデルビュー行列）}}
\begin{DoxyItemize}
\item モデリング変換（オブジェクトの平行移動、回転、スケーリング）と視野変換（カメラの視点設定）を行う行列。
\item 一般的な3\+Dシーンでは、この行列が頻繁に操作されます。
\end{DoxyItemize}
\item {\bfseries{{\ttfamily GL\+\_\+\+PROJECTION}（投影行列）}}
\begin{DoxyItemize}
\item 投影変換（3\+D空間を2\+Dスクリーンに投影する変換）を行う行列。
\item 透視投影（{\ttfamily glu\+Perspective}など）や平行投影（{\ttfamily gl\+Ortho}など）に使用されます。
\end{DoxyItemize}
\item {\bfseries{{\ttfamily GL\+\_\+\+TEXTURE}（テクスチャ行列）}}
\begin{DoxyItemize}
\item テクスチャ座標の変換を行う行列。
\item テクスチャを拡大・縮小したり、回転させたりする際に使用されます。
\end{DoxyItemize}
\end{DoxyEnumerate}

\begin{DoxyEnumerate}
\item {\bfseries{モダン\+Open\+GLでは使用されない}}
\begin{DoxyItemize}

\begin{DoxyItemize}

\end{DoxyItemize}
\end{DoxyEnumerate}
