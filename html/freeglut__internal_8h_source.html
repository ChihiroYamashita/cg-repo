492/*
493 * SET_WCB() is used as:
494 *
495 * SET_WCB( window, cbname, func );
496 *
497 * ...where {window} is the freeglut window to set the callback,
498 * {cbname} is the window-specific callback to set,
499 * {func} is a function-pointer.
500 *
501 * Originally, {FETCH_WCB( ... ) = func} was rather sloppily used,
502 * but this can cause warnings because the FETCH_WCB() macro type-
503 * casts its result, and a type-cast value shouldn't be an lvalue.
504 *
505 * The {if( FETCH_WCB( ... ) != func )} test is to do type-checking
506 * and for no other reason. Since it's hidden in the macro, the
507 * ugliness is felt to be rather benign.
508 */
509#define SET_WCB(window,cbname,func) \
510do \
511{ \
512 if( FETCH_WCB( window, cbname ) != (SFG_Proc)(func) ) \
513 (((window).CallBacks[CB_ ## cbname]) = (SFG_Proc)(func)); \
514} while( 0 )
515
516/*
517 * FETCH_WCB() is used as:
518 *
519 * FETCH_WCB( window, cbname );
520 *
521 * ...where {window} is the freeglut window to fetch the callback from,
522 * {cbname} is the window-specific callback to fetch.
523 *
524 * The result is correctly type-cast to the callback function pointer
525 * type.
526 */
527#define FETCH_WCB(window,cbname) \
528 ((window).CallBacks[CB_ ## cbname])
529
530/*
531 * INVOKE_WCB() is used as:
532 *
533 * INVOKE_WCB( window, cbname, ( arg_list ) );
534 *
535 * ...where {window} is the freeglut window,
536 * {cbname} is the window-specific callback to be invoked,
537 * {(arg_list)} is the parameter list.
538 *
539 * The callback is invoked as:
540 *
541 * callback( arg_list );
542 *
543 * ...so the parentheses are REQUIRED in the {arg_list}.
544 *
545 * NOTE that it does a sanity-check and also sets the
546 * current window.
547 *
548 */
549#if TARGET_HOST_MS_WINDOWS && !defined(_WIN32_WCE) /* FIXME: also WinCE? */
550#define INVOKE_WCB(window,cbname,arg_list) \
551do \
552{ \
553 if( FETCH_WCB( window, cbname ) ) \
554 { \
555 FGCB ## cbname func = (FGCB ## cbname)(FETCH_WCB( window, cbname )); \
556 fgSetWindow( &window ); \
557 func arg_list; \
558 } \
559} while( 0 )
560#else
561#define INVOKE_WCB(window,cbname,arg_list) \
562do \
563{ \
564 if( FETCH_WCB( window, cbname ) ) \
565 { \
566 fgSetWindow( &window ); \
567 ((FGCB ## cbname)FETCH_WCB( window, cbname )) arg_list; \
568 } \
569} while( 0 )
570#endif
