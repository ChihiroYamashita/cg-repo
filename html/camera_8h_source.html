100// rotate a given vector in_v around a given axis in_axis for a given angle in_angle_rad,
101// using quaternion
102
103// 入力 ( 回転させたいベクトル, 回転軸を表すベクトル,回転させる角度（ラジアン単位）)
104inline Eigen::Vector3d rotateVector( const Eigen::Vector3d& in_v, const Eigen::Vector3d& in_axis, const double& in_angle_rad )
105{
106 Eigen::Vector3d axis = in_axis.normalized();
107
108 const double cos_a = cos( in_angle_rad * 0.5 );
109 const double sin_a = sin( in_angle_rad * 0.5 );
110 const double m_sin_a = -sin_a;
111
112 // compute (r * p * q).v, where r, p, and q are quaternions given by
113 // p = (0, px, py, pz)
114 // q = (cos_a, axis_x * sin_a, axis_y * sin_a, axis_z * sin_a)
115 // r = (cos_a, axis_x * m_sin_a, axis_y * m_sin_a, axis_z * m_sin_a)
116
117 const double r_p_t = -m_sin_a * axis.dot( in_v );
118 const Eigen::Vector3d r_p = cos_a * in_v + m_sin_a * in_v.cross( axis );
119 return sin_a * r_p_t * axis + cos_a * r_p + sin_a * axis.cross( r_p );
120
121 //QQuaternion rotation = QQuaternion::fromAxisAndAngle(in_axis, qRadiansToDegrees(in_angle_rad));
122 //return rotation.rotatedVector(in_v);
123}
